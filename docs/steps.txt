this is how an nginx https config works:

    listen 443 ssl; # managed by Certbot
    listen [::]:443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/spyderfly.sizablesplash.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/spyderfly.sizablesplash.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
